cmake_minimum_required(VERSION 3.8)
project(bball_sim_host CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(unitree_hg REQUIRED)
# MUJOCO_ROOT overrides the conda libmujoco (3.3.6, the stock simulator's version).
if(MUJOCO_ROOT)
  set(MJ_INC ${MUJOCO_ROOT}/include)
  file(GLOB MJ_LIB ${MUJOCO_ROOT}/libmujoco.so*)
else()
  find_path(MJ_INC mujoco/mujoco.h REQUIRED)
  find_library(MJ_LIB mujoco REQUIRED)
endif()
find_library(EGL_LIB EGL REQUIRED)
find_library(OPENGL_LIB GL REQUIRED)
set(LODEPNG ${CMAKE_CURRENT_SOURCE_DIR}/../../../unitree_mujoco/simulate/src/lodepng)
add_executable(sim_host src/sim_host.cpp ${LODEPNG}/lodepng.cpp)
target_include_directories(sim_host PRIVATE ${MJ_INC} ${LODEPNG})
target_link_libraries(sim_host ${MJ_LIB} ${EGL_LIB} ${OPENGL_LIB})
ament_target_dependencies(sim_host rclcpp sensor_msgs unitree_hg)
set_target_properties(sim_host PROPERTIES BUILD_RPATH_USE_ORIGIN OFF INSTALL_RPATH "")
install(TARGETS sim_host DESTINATION lib/${PROJECT_NAME})
ament_package()
